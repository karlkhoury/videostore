pipeline {
    agent any

    triggers {
        // Poll GitHub every 2 minutes
        pollSCM('H/2 * * * *')
    }

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/karlkhoury/videostore'
            }
        }

        stage('Build in Minikube Docker') {
            steps {
                bat '''
                REM === Switch Docker to Minikube Docker ===
                call minikube docker-env --shell=cmd > docker_env.bat
                call docker_env.bat
                REM === Build Django image inside Minikube Docker ===
                docker build -t mydjangoapp:latest ./videostore
                '''
            }
        }

        stage('Deploy to Minikube') {
            steps {
                bat '''
		        REM === Use the correct Minikube kubeconfig path ===
                set KUBECONFIG=C:\\Users\\karl\\.kube\\config
                
		        REM === Confirm the Minikube context exists ===
                kubectl config use-context minikube
    
                REM === Apply manifests ===
                kubectl apply -f videostore/deployment.yaml
                kubectl apply -f videostore/service.yaml

                REM === Wait for rollout ===
                kubectl rollout status deployment/django-deployment
                '''
            }
        }
    }
}
